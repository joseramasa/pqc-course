\chapter*{Notation and Conventions}
\addcontentsline{toc}{chapter}{Notation and Conventions}

\begin{tabular}{lp{9.5cm}}
\hline\noalign{\smallskip}
\textbf{Symbol} & \textbf{Meaning} \\
\noalign{\smallskip}\svhline\noalign{\smallskip}

% Number sets
$\N, \Z, \Q, \R, \C$ & Natural, integer, rational, real, complex numbers \\
$\F_q$ & Finite field with $q$ elements \\
$\Zq$ & Integers modulo $q$ \\
$\Rq$ & Polynomial ring $\Zq[x]/(x^n + 1)$ \\

\noalign{\smallskip}\svhline\noalign{\smallskip}
% Linear algebra
$\mathbf{a}, \mathbf{b}$ & Column vectors (bold lowercase) \\
$\mathbf{A}, \mathbf{B}$ & Matrices (bold uppercase) \\
$\norm{\mathbf{v}}$ & Euclidean norm of $\mathbf{v}$ \\
$\ip{\mathbf{u}}{\mathbf{v}}$ & Inner product of $\mathbf{u}$ and $\mathbf{v}$ \\
$\mathbf{A}^\top$ & Matrix transpose \\

\noalign{\smallskip}\svhline\noalign{\smallskip}
% Probability
$x \getsr S$ & $x$ sampled uniformly at random from set $S$ \\
$x \leftarrow D$ & $x$ sampled according to distribution $D$ \\
$D_{\Z,\sigma}$ & Discrete Gaussian over $\Z$ with parameter $\sigma$ \\
$U(\Zq^n)$ & Uniform distribution over $\Zq^n$ \\
$\negl(\lambda)$ & Negligible function in security parameter $\lambda$ \\

\noalign{\smallskip}\svhline\noalign{\smallskip}
% Lattices
$\lat(\mathbf{B})$ & Lattice generated by basis $\mathbf{B}$ \\
$\lat^*$ & Dual lattice \\
$\lambda_i(\lat)$ & $i$-th successive minimum of $\lat$ \\
$\det(\lat)$ & Determinant of lattice $\lat$ \\

\noalign{\smallskip}\svhline\noalign{\smallskip}
% Cryptography
$\lambda$ & Security parameter \\
$\KeyGen, \Enc, \Dec$ & Key generation, encryption, decryption \\
$\Sign, \Verify$ & Signature generation, verification \\
$\Encaps, \Decaps$ & Key encapsulation, decapsulation \\
$\mathsf{pk}, \mathsf{sk}$ & Public key, secret key \\
$H, \mathcal{H}$ & Hash function, hash function family \\

\noalign{\smallskip}\svhline\noalign{\smallskip}
% Complexity
$\bigO(f(n))$ & Asymptotic upper bound \\
$\poly(n)$ & Some polynomial in $n$ \\

\noalign{\smallskip}\hline
\end{tabular}
