\chapter{Code-Based Cryptography}
\label{ch:code-based}

\abstract*{Code-based cryptography, rooted in the theory of error-correcting codes, offers the longest track record of any post-quantum approach: the McEliece cryptosystem has resisted cryptanalysis since 1978. This chapter develops the algebraic coding theory needed to understand Classic McEliece (a NIST candidate) and the structured-code schemes HQC and BIKE, connecting throughout to the quantum error correction and information theory familiar to physicists.}

\begin{prob}
\label{prob:ch09-hamming}
Construct the $[7,4,3]$ Hamming code. Write its generator matrix $G$ and parity-check matrix $H$. Encode the message $\mathbf{m} = (1,0,1,1)$, introduce a single-bit error, and decode using syndrome decoding.
\end{prob}

\begin{prob}
\label{prob:ch09-mceliece}
In the McEliece cryptosystem with a $[1024, 524, 101]$ Goppa code, compute the public key size in bytes. Compare this with the public key sizes of ML-KEM-768 and RSA-2048. Discuss the practical implications.
\end{prob}

\begin{prob}
\label{prob:ch09-isd}
Describe Prange's Information Set Decoding algorithm. Estimate the expected number of iterations to decode a random $[n, k, t]$ code. For $n = 2048$, $k = 1024$, $t = 50$, compute the work factor and compare with the security claims of Classic McEliece.
\end{prob}
