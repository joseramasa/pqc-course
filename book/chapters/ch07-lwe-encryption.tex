\chapter{Learning With Errors and Lattice-Based Encryption}
\label{ch:lwe-encryption}

\abstract*{The Learning With Errors (LWE) problem is the computational foundation of the most widely deployed post-quantum schemes. This chapter develops LWE from its definition through its hardness guarantees, introduces the structured variants Ring-LWE and Module-LWE that enable practical efficiency, and culminates in a complete description of ML-KEM (CRYSTALS-Kyber)---the NIST standard for post-quantum key encapsulation.}

\begin{prob}
\label{prob:ch07-lwe}
Consider LWE with $n = 4$, $q = 23$, and uniform ternary errors $e_i \in \{-1, 0, 1\}$. Given the secret $\mathbf{s} = (3, 7, 11, 15)^\top$ and random vectors $\mathbf{a}_i$, construct 6 LWE samples $(\mathbf{a}_i, b_i = \ip{\mathbf{a}_i}{\mathbf{s}} + e_i \bmod q)$. Verify that recovering $\mathbf{s}$ from these samples is non-trivial even at this small scale.
\end{prob}

\begin{prob}
\label{prob:ch07-ntt}
Implement the Number Theoretic Transform for $n = 8$ and $q = 17$. Verify that polynomial multiplication via NTT matches na\"ive multiplication in $\Zq[x]/(x^8 + 1)$. What are the conditions on $q$ for the NTT to exist?
\end{prob}

\begin{prob}
\label{prob:ch07-fo}
Explain why a CPA-secure encryption scheme is insufficient for use as a KEM in practice. Describe the Fujisaki-Okamoto transform and explain how it achieves CCA2 security. What is the role of the re-encryption check?
\end{prob}
